1. Implementation of symbol table.
----------------------------------------------
#include <stdio.h>
#include <string.h>
#define MAX 100
struct Symbol {
char name[20];
char type[10];
int size;
int line;
};
struct Symbol table[MAX];
int count = 0;
// Insert a symbol
void insert() {
char name[20], type[10];
int size, line;
printf("Enter symbol name: ");
scanf("%s", name);
// Check if symbol already exists
for (int i = 0; i < count; i++) {
if (strcmp(name, table[i].name) == 0) {
printf("Symbol already exists!\n");
return;
}
}
printf("Enter symbol type: ");
scanf("%s", type);
printf("Enter size: ");
scanf("%d", &size);
printf("Enter line number: ");
scanf("%d", &line);
strcpy(table[count].name, name);
strcpy(table[count].type, type);
table[count].size = size;
table[count].line = line;
count++;
printf("Symbol inserted successfully.\n");
}


// Display all symbols
void display() {
if (count == 0) {
printf("Symbol table is empty.\n");
return;
}
printf("\n%-20s %-10s %-5s %-5s\n", "Name", "Type", "Size", "Line");
printf("---------------------------------------------------\n");
for (int i = 0; i < count; i++) {
printf("%-20s %-10s %-5d %-5d\n", table[i].name, table[i].type, table[i].size,
table[i].line);
}
}
// Search for a symbol
void search() {
char name[20];
printf("Enter symbol name to search: ");
scanf("%s", name);
for (int i = 0; i < count; i++) {
if (strcmp(name, table[i].name) == 0) {
printf("Symbol found at position %d\n", i + 1);
printf("Name: %s, Type: %s, Size: %d, Line: %d\n",
table[i].name, table[i].type, table[i].size, table[i].line);
return;
}
}
printf("Symbol not found.\n");
}
// Modify a symbol
void modify() {
char name[20];
printf("Enter symbol name to modify: ");
scanf("%s", name);
for (int i = 0; i < count; i++) {
if (strcmp(name, table[i].name) == 0) {
printf("Enter new type: ");
scanf("%s", table[i].type);
printf("Enter new size: ");
scanf("%d", &table[i].size);
printf("Enter new line number: ");


scanf("%d", &table[i].line);
printf("Symbol updated successfully.\n");
return;
}
}
printf("Symbol not found.\n");
}
// Main menu
int main() {
int choice;
while (1) {
printf("\n----- Symbol Table Menu -----\n");
printf("1. Insert\n");
printf("2. Display\n");
printf("3. Search\n");
printf("4. Modify\n");
printf("5. Exit\n");
printf("Enter your choice: ");
scanf("%d", &choice);
switch (choice) {
case 1: insert(); break;
case 2: display(); break;
case 3: search(); break;
case 4: modify(); break;
case 5: return 0;
default: printf("Invalid choice!\n");
}
}
return 0;
}

Implement Any One Storage Allocation Strategy (Heap, Stack, or Static)
Implementation of Stack-Based Storage Allocation Strategy
#include <stdio.h>
#include <string.h>
#define SIZE 10
typedef struct {
char name[10];
int value;
} Var;
Var stack[SIZE];
int top = -1;
// Function to push a variable onto the stack
void push(char name[], int value) {
if (top < SIZE - 1) {
top++;
strcpy(stack[top].name, name);
stack[top].value = value;
printf("Allocated %s = %d\n", name, value);
} else {
printf("Stack Overflow: Cannot allocate %s\n", name);
}
}
// Function to pop a variable from the stack


void pop() {
if (top >= 0) {
printf("Deallocated %s\n", stack[top].name);
top--;
} else {
printf("Stack Underflow: No variable to deallocate\n");
}
}
// Function to display current stack content
void display() {
printf("Stack:\n");
for (int i = top; i >= 0; i--)
printf(" %s = %d\n", stack[i].name, stack[i].value);
}
int main() {
push("a", 10);
push("b", 20);
display();
pop();
display();
return 0;
}

lookahead = *++input;

void E(); // E -> T E'
void E_(); // E' -> + T E' | ε
void T(); // T -> F T'
void T_(); // T' -> * F T' | ε
void F(); // F -> ( E ) | id


void E() {
T();
E_();
}
void E_() {
if (lookahead == '+') {
match('+');
T();
E_();
}
// else epsilon (do nothing)
}
void T() {
F();
T_();
}
void T_() {
if (lookahead == '*') {
match('*');
F();
T_();
}
// else epsilon
}
void F() {
if (lookahead == '(') {


match('(');
E();
match(')');
} else if (isalpha(lookahead)) {
match(lookahead); // id is a single letter like 'a', 'x', etc.

if (!strcmp(arr[i], sym)) return true;
return false;
}

for (int i = 0; i < scount; i++)
if (!exists(target, *count, source[i]))
strcpy(target[(*count)++], source[i]);
}
